int getValidatedChoice(int min, int max) {
    int choice;
    while (true) {
        cout << "Enter choice: ";
        if (cin >> choice && choice >= min && choice <= max) {
            cin.ignore(10000, '\n');
            return choice;
        }
        else {
            cout << "[error] Invalid input. Please enter a number between " << min << " and " << max << "." << endl;
            cin.clear();
            cin.ignore(10000, '\n');
        }
    }
}
